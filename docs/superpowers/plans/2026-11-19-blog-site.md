# Blog (Site Side) Implementation Plan — Plan 1 of 2

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the blog into lidr.io: a build-time content pipeline (validate → render → optimise images → OG images), the `/blog` pages, SEO, footer link and tests. Everything runs against local fixture content.

**Architecture:** A `tsx` prebuild script (`scripts/blog/prepare.ts`) reads Markdown content from a directory (`BLOG_CONTENT_DIR`) or a cloned repo (`BLOG_CONTENT_REPO`). It does all the heavy work and writes one JSON file, `.blog/data.json`, plus generated images under `public/images/blog/`. Next.js pages only read that JSON, so the presentation layer stays thin. A postbuild script (`scripts/blog/finalize.ts`) removes placeholder routes and appends redirects to `out/_redirects`. With neither env var set, the blog is "disabled": an empty, noindexed `/blog/` and no footer link. That is the production state until Plan 2 creates the content repo.

**Tech Stack:** Next.js 16.2 (`output: "export"`), TypeScript, tsx, unified/remark/rehype, rehype-sanitize, gray-matter, yaml, sharp, exif-reader, satori, wawoff2, vitest, Playwright, axe, Lighthouse CI.

**Spec:** `docs/superpowers/specs/2026-11-19-blog-design.md` (rev 2). Plan 2 (`2026-11-19-blog-publishing.md`, written after this lands) covers the content repo, CMS, auth Worker, Actions, Claude skill and docs.

**Deviations from the spec, decided while planning (all simplifications):**
- **Verified:** `generateStaticParams` returning `[]` fails `next build` under `output: export` ("missing generateStaticParams()"). The `__empty` placeholder approach (spec §4) is required and was verified to build. `app/.../rss.xml/route.ts` with `dynamic = "force-static"` emits `out/.../rss.xml` correctly with `trailingSlash: true`.
- **Validation is a hand-written rules module (`lib/blog/rules.ts` + `validate.ts`), not zod.** Every field needs a hard/soft decision and a fallback, which zod doesn't model. `RULES` stays the single source Plan 2 uses to generate the CMS config.
- **HEIC is not accepted.** sharp's prebuilt binaries don't decode HEIC (HEVC). HEIC files get a soft "export as JPEG" problem (spec §12 item 5 fallback).
- **No image cache** (YAGNI): the image count is small. Add one later if build time grows.
- **OG images are generated in prebuild with satori** (spec §6), not `opengraph-image.tsx`.

**Branch:** do this work on `feat/blog-site` (pushing `main` deploys production).

import matter from "gray-matter";
import { parse as parseYaml } from "yaml";
import { RULES } from "./rules";
import type { PostMeta, Problem, TagDef } from "./types";

const hard = (field: string, message: string): Problem => ({ level: "hard", field, message });
const soft = (field: string, message: string): Problem => ({ level: "soft", field, message });

export type ParsedFile =
  | { ok: true; data: Record<string, unknown>; body: string }
  | { ok: false; problem: Problem };

export function parsePostFile(raw: string): ParsedFile {
  try {
    // Parse front-matter with the `yaml` package (YAML 1.2, safe by default)
    // instead of gray-matter's bundled js-yaml 3.x, which has a known
    // prototype-pollution advisory. Passing options also disables
    // gray-matter's cache (we parse historical versions of the same file).
    const { data, content } = matter(raw, {
      engines: { yaml: (s: string) => (parseYaml(s) ?? {}) as object },
    });
    return { ok: true, data: data as Record<string, unknown>, body: content.trim() };
  } catch (err) {
    return {
      ok: false,
      problem: hard(
        "front-matter",
        `The post's settings block can't be read (${(err as Error).message.split("\n")[0]}). Open it in the editor and save it again.`
      ),
    };
  }
}

function toDateString(v: unknown): string | null {
  if (v instanceof Date && !Number.isNaN(v.getTime())) return v.toISOString().slice(0, 10);
  if (typeof v === "string" && RULES.datePattern.test(v) && !Number.isNaN(Date.parse(v))) return v;
  return null;
}

export function excerpt(text: string, max: number = RULES.excerptLength): string {
  const t = text.replace(/\s+/g, " ").trim();
  if (t.length <= max) return t;
  const cut = t.lastIndexOf(" ", max);
  return `${t.slice(0, cut > 0 ? cut : max).trim()}…`;
}

export function validatePost(input: {
  slug: string;
  data: Record<string, unknown>;
  body: string;
  plainText: string;
  tags: Map<string, TagDef>;
}): { meta: PostMeta | null; problems: Problem[] } {
  const { slug, data } = input;
  const p: Problem[] = [];

  if (!RULES.slugPattern.test(slug)) {
    p.push(hard("slug", `The file name "${slug}" must use only lowercase letters, numbers and hyphens.`));
  }
  const title = typeof data.title === "string" ? data.title.trim() : "";
  if (!title) p.push(hard("title", "The post has no title."));
  const date = toDateString(data.date);
  if (!date) p.push(hard("date", "The publish date is missing or isn't a real date (YYYY-MM-DD)."));
  if (!input.body.trim()) p.push(hard("body", "The post has no text."));
  if (p.length > 0 || !date) return { meta: null, problems: p };

  const { title: tr, description: dr, tags: tg } = RULES;
  if (title.length < tr.min || title.length > tr.max) {
    p.push(soft("title", `The title is ${title.length} characters. Aim for ${tr.min}–${tr.max} so Google shows it in full.`));
  }

  let description = typeof data.description === "string" ? data.description.trim() : "";
  if (!description) {
    p.push(soft("description", `The description is empty, so the opening text is used instead. Write one (${dr.min}–${dr.max} characters) for Google results.`));
    description = excerpt(input.plainText);
  } else if (description.length > dr.max) {
    p.push(soft("description", `The description is ${description.length} characters. Google cuts it off after about ${dr.max}. Shorten it.`));
  } else if (description.length < dr.min) {
    p.push(soft("description", `The description is only ${description.length} characters. Aim for ${dr.min}–${dr.max} so Google has enough to show.`));
  }

  let updated: string | undefined;
  if (data.updated !== undefined && data.updated !== null && data.updated !== "") {
    const u = toDateString(data.updated);
    if (!u || u < date) {
      p.push(soft("updated", "The 'updated' date isn't valid or is earlier than the publish date, so it was ignored."));
    } else updated = u;
  }

  const rawTags = Array.isArray(data.tags) ? data.tags.filter((t): t is string => typeof t === "string") : [];
  const unknown = rawTags.filter((t) => !input.tags.has(t));
  let tags = rawTags.filter((t) => input.tags.has(t));
  if (unknown.length) {
    p.push(soft("tags", `These tags don't exist and were removed: ${unknown.join(", ")}. Pick from the tag list.`));
  }
  if (tags.length < tg.min) {
    p.push(soft("tags", "The post has no tags. Add 1–3 so readers can find related posts."));
  } else if (tags.length > tg.max) {
    p.push(soft("tags", `The post has ${tags.length} tags. Only the first ${tg.max} are used.`));
    tags = tags.slice(0, tg.max);
  }

  const cover = typeof data.cover === "string" && data.cover.trim() ? data.cover.trim() : undefined;
  let coverAlt = typeof data.coverAlt === "string" ? data.coverAlt.trim() : "";
  if (cover && !coverAlt) {
    p.push(soft("coverAlt", "The cover photo has no description (alt text), so the title is used. Describe what the photo shows."));
    coverAlt = title;
  }

  let redirectTo: string | undefined;
  if (data.redirectTo !== undefined && data.redirectTo !== "") {
    if (typeof data.redirectTo === "string" && data.redirectTo.startsWith("/") && !data.redirectTo.startsWith("//")) {
      redirectTo = data.redirectTo;
    } else p.push(soft("redirectTo", "The redirect must be a path on lidr.io starting with /. It was ignored."));
  }

  let canonical: string | undefined;
  if (data.canonical !== undefined && data.canonical !== "") {
    if (typeof data.canonical === "string" && data.canonical.startsWith("https://")) canonical = data.canonical;
    else p.push(soft("canonical", "The canonical URL must start with https://. It was ignored."));
  }

  return {
    meta: {
      slug,
      title,
      description,
      date,
      updated,
      tags,
      cover,
      coverAlt: cover ? coverAlt : undefined,
      draft: data.draft === true,
      redirectTo,
      canonical,
    },
    problems: p,
  };
}

export function parseTags(yamlText: string): { tags: Map<string, TagDef>; problems: Problem[] } {
  const tags = new Map<string, TagDef>();
  const problems: Problem[] = [];
  let doc: unknown;
  try {
    doc = parseYaml(yamlText);
  } catch {
    return { tags, problems: [soft("tags.yml", "The tag list can't be read. Open Tags in the editor and save it again.")] };
  }
  const list = (doc as { tags?: unknown })?.tags;
  if (!Array.isArray(list)) return { tags, problems };
  for (const entry of list) {
    const e = entry as Partial<TagDef>;
    if (typeof e?.slug === "string" && RULES.slugPattern.test(e.slug) && typeof e.name === "string" && e.name.trim()) {
      tags.set(e.slug, { slug: e.slug, name: e.name.trim(), description: typeof e.description === "string" ? e.description.trim() : "" });
    } else {
      problems.push(soft("tags.yml", `A tag is missing a name or has an invalid slug (${JSON.stringify(entry)}). It was skipped.`));
    }
  }
  return { tags, problems };
}

import type { Element, Root } from "hast";
import { toString as mdToString } from "mdast-util-to-string";
import rehypeAutolinkHeadings from "rehype-autolink-headings";
import rehypeSanitize from "rehype-sanitize";
import rehypeSlug from "rehype-slug";
import rehypeStringify from "rehype-stringify";
import remarkGfm from "remark-gfm";
import remarkParse from "remark-parse";
import remarkRehype from "remark-rehype";
import { unified } from "unified";
import { SKIP, visit } from "unist-util-visit";
import type { Problem } from "./types";

export type ImageVariant = { src: string; srcset: string; width: number; height: number };
export type ImageManifest = Record<string, ImageVariant>;

/** Body column is 728px; images span it on desktop, full width on mobile. */
const BODY_IMAGE_SIZES = "(min-width: 760px) 728px, 100vw";

const soft = (field: string, message: string): Problem => ({ level: "soft", field, message });

export function markdownToPlainText(md: string): string {
  const tree = unified().use(remarkParse).use(remarkGfm).parse(md);
  // Join block-level text with spaces, then normalise whitespace.
  const blocks = (tree.children ?? []).map((n) => mdToString(n));
  return blocks.join(" ").replace(/\s+/g, " ").trim();
}

function rehypeBlogImages(opts: { images: ImageManifest; warnings: Problem[] }) {
  return (tree: Root) => {
    let first = true;
    visit(tree, "element", (node: Element, index, parent) => {
      if (node.tagName !== "img" || !parent || index === undefined) return;
      const src = String(node.properties.src ?? "");
      const alt = String(node.properties.alt ?? "");
      if (!alt) {
        opts.warnings.push(soft("images", `A photo has no alt text (${src}). Describe what it shows so screen readers and Google understand it.`));
      }
      if (!src.startsWith("/images/blog/")) {
        node.properties.loading = "lazy";
        return;
      }
      const img = opts.images[src];
      if (!img) {
        opts.warnings.push(soft("images", `Photo not found: ${src}. Upload it again in the editor.`));
        parent.children.splice(index, 1);
        return [SKIP, index];
      }
      const picture: Element = {
        type: "element",
        tagName: "picture",
        properties: {},
        children: [
          { type: "element", tagName: "source", properties: { type: "image/webp", srcSet: img.srcset, sizes: BODY_IMAGE_SIZES }, children: [] },
          {
            type: "element",
            tagName: "img",
            properties: {
              src: img.src,
              alt,
              width: img.width,
              height: img.height,
              ...(first ? {} : { loading: "lazy" }),
              decoding: "async",
            },
            children: [],
          },
        ],
      };
      first = false;
      parent.children[index] = picture;
      return SKIP;
    });
  };
}

function rehypeExternalLinks() {
  return (tree: Root) => {
    visit(tree, "element", (node: Element) => {
      if (node.tagName !== "a") return;
      const href = String(node.properties.href ?? "");
      if (/^https?:\/\//.test(href) && !/^https:\/\/(www\.)?lidr\.io(\/|$)/.test(href)) {
        node.properties.rel = ["noopener", "noreferrer"];
      }
    });
  };
}

function rehypeWrapTables() {
  return (tree: Root) => {
    visit(tree, "element", (node: Element, index, parent) => {
      if (node.tagName !== "table" || !parent || index === undefined) return;
      parent.children[index] = { type: "element", tagName: "div", properties: { className: ["table-wrap"] }, children: [node] };
      return SKIP;
    });
  };
}

export async function renderMarkdown(
  md: string,
  opts: { images: ImageManifest }
): Promise<{ html: string; words: number; warnings: Problem[] }> {
  const warnings: Problem[] = [];
  const file = await unified()
    .use(remarkParse)
    .use(remarkGfm)
    .use(remarkRehype) // raw HTML is dropped (allowDangerousHtml is off)
    .use(rehypeSanitize) // default GitHub schema: no scripts, handlers, javascript: URLs
    .use(rehypeSlug)
    .use(rehypeAutolinkHeadings, { behavior: "wrap" })
    .use(rehypeBlogImages, { images: opts.images, warnings })
    .use(rehypeExternalLinks)
    .use(rehypeWrapTables)
    .use(rehypeStringify)
    .process(md);
  const text = markdownToPlainText(md);
  return { html: String(file), words: text ? text.split(/\s+/).length : 0, warnings };
}

type Relatable = { slug: string; date: string; tags: string[] };

/** For each post: up to 3 other posts sharing ≥1 tag, most shared first, then newest. */
export function computeRelated(posts: Relatable[], limit = 3): Map<string, string[]> {
  const out = new Map<string, string[]>();
  for (const post of posts) {
    const own = new Set(post.tags);
    const scored = posts
      .filter((o) => o.slug !== post.slug)
      .map((o) => ({ o, shared: o.tags.filter((t) => own.has(t)).length }))
      .filter((s) => s.shared > 0)
      .sort((a, b) => b.shared - a.shared || b.o.date.localeCompare(a.o.date) || a.o.slug.localeCompare(b.o.slug));
    out.set(post.slug, scored.slice(0, limit).map((s) => s.o.slug));
  }
  return out;
}

/** Page 1 shows 1 featured post + 9 cards; later pages show 9 cards. */
export const FIRST_PAGE_SIZE = 10;
export const PAGE_SIZE = 9;

export function pageCount(total: number): number {
  return total <= FIRST_PAGE_SIZE ? 1 : 1 + Math.ceil((total - FIRST_PAGE_SIZE) / PAGE_SIZE);
}

export function pageSlice<T>(items: T[], page: number): { featured?: T; items: T[] } {
  if (page === 1) return { featured: items[0], items: items.slice(1, FIRST_PAGE_SIZE) };
  const start = FIRST_PAGE_SIZE + (page - 2) * PAGE_SIZE;
  return { items: items.slice(start, start + PAGE_SIZE) };
}

import { existsSync, mkdirSync, readdirSync, readFileSync, statSync } from "node:fs";
import { basename, extname, join } from "node:path";
import exifReader from "exif-reader";
import sharp from "sharp";
import type { ImageManifest } from "../../lib/blog/markdown";
import { RULES } from "../../lib/blog/rules";
import type { Problem } from "../../lib/blog/types";

export const WIDTHS = [480, 960, 1600] as const;

const soft = (message: string): Problem => ({ level: "soft", field: "images", message });

function hasGps(exif: Buffer | undefined): boolean {
  if (!exif) return false;
  try {
    const info = exifReader(exif).GPSInfo;
    return !!info && Object.keys(info).length > 0;
  } catch {
    return false;
  }
}

/**
 * Turns content/images/<slug>/<file> into optimised, metadata-free files at
 * public/images/blog/<slug>/<name>-<w>.webp and <name>-1600.jpg.
 * Manifest keys are the paths authors use in Markdown: /images/blog/<slug>/<file>.
 */
export async function processImages(opts: {
  contentDir: string;
  publicDir: string;
}): Promise<{ manifest: ImageManifest; problems: Record<string, Problem[]> }> {
  const manifest: ImageManifest = {};
  const problems: Record<string, Problem[]> = {};
  const add = (slug: string, p: Problem) => (problems[slug] ??= []).push(p);
  const root = join(opts.contentDir, "images");
  if (!existsSync(root)) return { manifest, problems };

  for (const slug of readdirSync(root).sort()) {
    const dir = join(root, slug);
    if (!statSync(dir).isDirectory()) continue;
    for (const file of readdirSync(dir).sort()) {
      if (file.startsWith(".")) continue;
      const ext = extname(file).slice(1).toLowerCase();
      if (!(RULES.imageExtensions as readonly string[]).includes(ext)) {
        add(slug, soft(`"${file}" is a ${ext.toUpperCase() || "unknown"} file, which browsers can't show. Save it as JPEG and upload it again.`));
        continue;
      }
      const input = readFileSync(join(dir, file));
      if (input.length > RULES.maxImageBytes) {
        add(slug, soft(`"${file}" is larger than 15 MB and was skipped. Upload a smaller version.`));
        continue;
      }
      let meta: sharp.Metadata;
      try {
        meta = await sharp(input).metadata();
      } catch {
        add(slug, soft(`"${file}" can't be opened as an image. Upload it again.`));
        continue;
      }
      if (hasGps(meta.exif)) {
        add(slug, soft(`"${file}" contains the location where it was taken. The published copy has it removed, but the original upload is public. Turn off location before uploading photos.`));
      }

      const name = basename(file, extname(file));
      const outDir = join(opts.publicDir, "images", "blog", slug);
      mkdirSync(outDir, { recursive: true });
      const urlBase = `/images/blog/${slug}/${name}`;

      const srcset: string[] = [];
      const seen = new Set<number>();
      let width = 0;
      let height = 0;
      for (const w of WIDTHS) {
        const info = await sharp(input)
          .rotate() // apply EXIF orientation; output has no metadata
          .resize({ width: w, withoutEnlargement: true })
          .webp({ quality: 78 })
          .toFile(join(outDir, `${name}-${w}.webp`));
        if (!seen.has(info.width)) {
          seen.add(info.width);
          srcset.push(`${urlBase}-${w}.webp ${info.width}w`);
        }
        width = info.width;
        height = info.height;
      }
      await sharp(input)
        .rotate()
        .resize({ width: 1600, withoutEnlargement: true })
        .jpeg({ quality: 80, mozjpeg: true })
        .toFile(join(outDir, `${name}-1600.jpg`));

      manifest[`/images/blog/${slug}/${file}`] = { src: `${urlBase}-1600.jpg`, srcset: srcset.join(", "), width, height };
    }
  }
  return { manifest, problems };
}

import { join } from "node:path";
import type { ReactNode } from "react";
import satori from "satori";
import sharp from "sharp";
import { decompress } from "wawoff2";

// Colours mirror styles/tokens.css (--color-brand, --color-accent, --color-white).
const BRAND = "#192c27";
const ACCENT = "#f2e32b";
const WHITE = "#ffffff";

let fontData: Buffer | null = null;
async function amenti(): Promise<Buffer> {
  // satori can't read woff2; decompress the site's own font to TTF in memory.
  fontData ??= Buffer.from(await decompress(readFileSync(join(process.cwd(), "app/fonts/Amenti-Medium.woff2"))));
  return fontData;
}

let logo: { uri: string; width: number; height: number } | null = null;
function footerLogo(height: number) {
  if (!logo) {
    const svg = readFileSync(join(process.cwd(), "public/icons/footer_logo.svg"), "utf8");
    const [, , w, h] = (svg.match(/viewBox="([^"]+)"/)?.[1] ?? "0 0 199 80").split(/\s+/).map(Number);
    logo = { uri: `data:image/svg+xml;base64,${Buffer.from(svg).toString("base64")}`, width: w, height: h };
  }
  return { src: logo.uri, height, width: Math.round((height * logo.width) / logo.height) };
}

type Node = { type: string; props: Record<string, unknown> };
const el = (type: string, style: Record<string, unknown>, children?: unknown, extra: Record<string, unknown> = {}): Node => ({
  type,
  props: { style, children, ...extra },
});

export async function renderOgImage(opts: { title: string; tagName?: string }): Promise<Buffer> {
  const len = opts.title.length;
  const fontSize = len <= 40 ? 72 : len <= 60 ? 60 : 52;
  const img = footerLogo(56);
  const tree = el(
    "div",
    {
      width: 1200,
      height: 630,
      display: "flex",
      flexDirection: "column",
      justifyContent: "space-between",
      background: BRAND,
      padding: "72px 80px",
      fontFamily: "Amenti",
      color: WHITE,
    },
    [
      el("div", { display: "flex", flexDirection: "column" }, [
        el("div", { display: "flex", fontSize, lineHeight: 1.15, maxHeight: Math.ceil(fontSize * 1.15 * 3), overflow: "hidden" }, opts.title),
        el("div", { width: 96, height: 8, background: ACCENT, marginTop: 32 }),
      ]),
      el("div", { display: "flex", justifyContent: "space-between", alignItems: "center" }, [
        el("img", {}, undefined, img),
        el("div", { display: "flex", fontSize: 28, color: ACCENT }, opts.tagName ?? ""),
      ]),
    ]
  );
  const svg = await satori(tree as unknown as ReactNode, {
    width: 1200,
    height: 630,
    fonts: [{ name: "Amenti", data: await amenti(), weight: 500, style: "normal" }],
  });
  return sharp(Buffer.from(svg)).png().toBuffer();
}

import { execFileSync } from "node:child_process";

const run = (cwd: string, args: string[]) =>
  execFileSync("git", args, { cwd, stdio: ["ignore", "pipe", "ignore"], maxBuffer: 20 * 1024 * 1024 }).toString();

/**
 * Walks posts/<slug>.md history newest → oldest and returns the first
 * version `isGood` accepts. Skips the working-tree version (assumed broken).
 * Returns null if the content dir isn't a git repo or nothing passes.
 */
export function findLastGood(contentDir: string, slug: string, isGood: (raw: string) => boolean): string | null {
  const path = `posts/${slug}.md`;
  let shas: string[];
  try {
    shas = run(contentDir, ["log", "--format=%H", "--", path]).split("\n").filter(Boolean);
  } catch {
    return null;
  }
  for (const sha of shas) {
    let raw: string;
    try {
      raw = run(contentDir, ["show", `${sha}:${path}`]);
    } catch {
      continue; // file deleted in that commit
    }
    if (isGood(raw)) return raw;
  }
  return null;
}

import { execFileSync } from "node:child_process";
import { existsSync, rmSync } from "node:fs";
import { resolve } from "node:path";

/**
 * Where blog content comes from:
 *   BLOG_CONTENT_DIR  — a local folder (tests, local editing)
 *   BLOG_CONTENT_REPO — a git URL, cloned fresh into `cloneTarget` (production)
 *   neither           — blog disabled (returns null)
 * A failed clone THROWS so the build fails and Cloudflare keeps the last
 * good deployment instead of publishing an empty blog.
 */
export function resolveContentDir(env: NodeJS.ProcessEnv | Record<string, string | undefined>, cloneTarget: string): string | null {
  if (env.BLOG_CONTENT_DIR) {
    const dir = resolve(env.BLOG_CONTENT_DIR);
    if (!existsSync(dir)) throw new Error(`BLOG_CONTENT_DIR ${dir} does not exist`);
    return dir;
  }
  if (env.BLOG_CONTENT_REPO) {
    rmSync(cloneTarget, { recursive: true, force: true });
    try {
      // blob:none keeps the clone small; history blobs load lazily for findLastGood.
      execFileSync("git", ["clone", "--quiet", "--filter=blob:none", env.BLOG_CONTENT_REPO, cloneTarget], { stdio: "pipe" });
    } catch (err) {
      throw new Error(`Could not fetch blog content from ${env.BLOG_CONTENT_REPO}: ${(err as Error).message}`);
    }
    return cloneTarget;
  }
  return null;
}

import { existsSync, mkdirSync, readdirSync, readFileSync, writeFileSync } from "node:fs";
import { basename, join } from "node:path";
import { markdownToPlainText, renderMarkdown } from "../../lib/blog/markdown";
import { computeRelated } from "../../lib/blog/related";
import { RULES } from "../../lib/blog/rules";
import { EMPTY_BLOG, type BlogData, type BlogReport, type BuiltPost, type PostMeta, type Problem, type TagDef } from "../../lib/blog/types";
import { parsePostFile, parseTags, validatePost } from "../../lib/blog/validate";
import { findLastGood } from "./history";
import { processImages } from "./images";
import { renderOgImage } from "./og";

function analyse(slug: string, raw: string, tags: Map<string, TagDef>): { meta: PostMeta | null; body: string; problems: Problem[] } {
  const parsed = parsePostFile(raw);
  if (!parsed.ok) return { meta: null, body: "", problems: [parsed.problem] };
  const plainText = markdownToPlainText(parsed.body);
  const r = validatePost({ slug, data: parsed.data, body: parsed.body, plainText, tags });
  return { ...r, body: parsed.body };
}

export async function buildBlogData(opts: {
  contentDir: string | null;
  publicDir: string;
  strict: boolean;
}): Promise<{ data: BlogData; report: BlogReport }> {
  const report: BlogReport = { problems: {} };
  if (!opts.contentDir) return { data: { ...EMPTY_BLOG }, report };
  const contentDir = opts.contentDir;
  const add = (key: string, ps: Problem[]) => {
    if (ps.length) (report.problems[key] ??= []).push(...ps);
  };

  const tagsFile = join(contentDir, "tags.yml");
  const { tags, problems: tagProblems } = existsSync(tagsFile)
    ? parseTags(readFileSync(tagsFile, "utf8"))
    : { tags: new Map<string, TagDef>(), problems: [] };
  add("_tags", tagProblems);

  const { manifest, problems: imageProblems } = await processImages({ contentDir, publicDir: opts.publicDir });
  for (const [slug, ps] of Object.entries(imageProblems)) add(slug, ps);

  const postsDir = join(contentDir, "posts");
  const files = existsSync(postsDir) ? readdirSync(postsDir).filter((f) => f.endsWith(".md")).sort() : [];
  const ogDir = join(opts.publicDir, "images", "blog", "og");
  mkdirSync(ogDir, { recursive: true });

  const built: Omit<BuiltPost, "related">[] = [];
  const redirects: BlogData["redirects"] = [];

  for (const file of files) {
    const slug = basename(file, ".md");
    let { meta, body, problems } = analyse(slug, readFileSync(join(postsDir, file), "utf8"), tags);
    let source: BuiltPost["source"] = "current";
    add(slug, problems);

    if (!meta) {
      const lastGood = findLastGood(contentDir, slug, (raw) => analyse(slug, raw, tags).meta !== null);
      if (!lastGood) continue;
      ({ meta, body } = analyse(slug, lastGood, tags));
      if (!meta) continue;
      source = "last-good";
      add(slug, [{ level: "soft", field: "post", message: "Your latest change has a problem, so the previous version is still live. Fix the problem above and publish again." }]);
    }

    if (meta.redirectTo) {
      redirects.push({ from: `/blog/${slug}/`, to: meta.redirectTo });
      continue;
    }
    if (meta.draft) continue;

    const rendered = await renderMarkdown(body, { images: manifest });
    add(slug, rendered.warnings);

    let cover: BuiltPost["cover"];
    if (meta.cover) {
      const img = manifest[meta.cover];
      if (img) cover = { ...img, alt: meta.coverAlt ?? meta.title };
      else add(slug, [{ level: "soft", field: "cover", message: `Cover photo not found: ${meta.cover}. Upload it again in the editor.` }]);
    }

    let ogImage = cover?.src;
    if (!ogImage) {
      const png = await renderOgImage({ title: meta.title, tagName: meta.tags[0] ? tags.get(meta.tags[0])?.name : undefined });
      writeFileSync(join(ogDir, `${slug}.png`), png);
      ogImage = `/images/blog/og/${slug}.png`;
    }

    built.push({
      slug,
      title: meta.title,
      description: meta.description,
      date: meta.date,
      updated: meta.updated,
      tags: meta.tags,
      cover,
      html: rendered.html,
      readingMinutes: Math.max(1, Math.ceil(rendered.words / RULES.wordsPerMinute)),
      canonical: meta.canonical,
      ogImage,
      source,
    });
  }

  built.sort((a, b) => b.date.localeCompare(a.date) || a.slug.localeCompare(b.slug));
  const related = computeRelated(built);
  const posts: BuiltPost[] = built.map((p) => ({ ...p, related: related.get(p.slug) ?? [] }));

  const counts = new Map<string, number>();
  posts.forEach((p) => p.tags.forEach((t) => counts.set(t, (counts.get(t) ?? 0) + 1)));
  const tagPages = [...tags.values()]
    .filter((t) => counts.has(t.slug))
    .map((t) => ({ ...t, count: counts.get(t.slug)! }))
    .sort((a, b) => a.slug.localeCompare(b.slug));

  if (opts.strict) {
    const total = Object.values(report.problems).flat().length;
    if (total > 0) {
      const lines = Object.entries(report.problems).flatMap(([k, ps]) => ps.map((p) => `  ${k} [${p.level}] ${p.field}: ${p.message}`));
      throw new Error(`Blog content has ${total} problem(s):\n${lines.join("\n")}`);
    }
  }

  return { data: { enabled: true, posts, tags: tagPages, redirects }, report };
}

import { appendFileSync, existsSync, readdirSync, readFileSync, rmSync } from "node:fs";
import { join } from "node:path";
import type { BlogData } from "../../lib/blog/types";

/** Placeholder param used when a dynamic blog route has nothing to build (see lib/blog/data.ts). */
const PLACEHOLDER = "__empty";

function removePlaceholders(dir: string) {
  for (const entry of readdirSync(dir, { withFileTypes: true })) {
    const path = join(dir, entry.name);
    if (entry.name.startsWith(PLACEHOLDER)) rmSync(path, { recursive: true, force: true });
    else if (entry.isDirectory()) removePlaceholders(path);
  }
}

export function finalizeOut(outDir: string, data: BlogData) {
  const blogDir = join(outDir, "blog");
  if (existsSync(blogDir)) removePlaceholders(blogDir);
  if (data.redirects.length) {
    const lines = data.redirects.flatMap(({ from, to }) => [`${from}  ${to}  301`, `${from.replace(/\/$/, "")}  ${to}  301`]);
    appendFileSync(join(outDir, "_redirects"), `\n# blog (generated)\n${lines.join("\n")}\n`);
  }

import { mkdirSync, mkdtempSync, rmSync, writeFileSync } from "node:fs";
import { tmpdir } from "node:os";
import { join } from "node:path";
import { buildBlogData } from "./pipeline";
import { resolveContentDir } from "./source";

/**
 * npm run blog:prepare — runs before `next build` / `next dev`.
 *   writes .blog/data.json (read by the Next pages) and .blog/report.json
 *   writes optimised images + OG images into public/images/blog/
 * npm run blog:check — same, strict, outputs to a temp dir (CI / Claude skill).
 */
async function main() {
  const cwd = process.cwd();
  const check = process.argv.includes("--check");
  const strict = process.env.BLOG_STRICT === "1";
  const contentDir = resolveContentDir(process.env, join(cwd, ".blog-content"));
  const publicDir = check ? mkdtempSync(join(tmpdir(), "blog-check-")) : join(cwd, "public");
  if (!check) rmSync(join(publicDir, "images", "blog"), { recursive: true, force: true });

  const { data, report } = await buildBlogData({ contentDir, publicDir, strict });

  if (!check) {
    mkdirSync(join(cwd, ".blog"), { recursive: true });
    writeFileSync(join(cwd, ".blog/data.json"), JSON.stringify(data));
    writeFileSync(join(cwd, ".blog/report.json"), JSON.stringify(report, null, 2));
  }

  const problems = Object.entries(report.problems);
  console.log(
    data.enabled
      ? `blog: ${data.posts.length} post(s), ${data.tags.length} tag(s), ${data.redirects.length} redirect(s), ${problems.length} item(s) with problems`
      : "blog: disabled (set BLOG_CONTENT_REPO or BLOG_CONTENT_DIR to enable)"
  );
  for (const [key, ps] of problems) for (const p of ps) console.warn(`  ${key} [${p.level}] ${p.field}: ${p.message}`);
}

main().catch((err) => {
  console.error(err instanceof Error ? err.message : err);
  process.exit(1);
});

const MONTHS = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];

/** "2026-11-01" → "1 Nov 2026". Parses the string directly: no Date, no timezone drift. */
export function formatDate(isoDate: string): string {
  const [y, m, d] = isoDate.split("-").map(Number);
  return `${d} ${MONTHS[m - 1]} ${y}`;

import type { BuiltPost } from "./types";

export const SITE = "https://lidr.io";
const ORG = { "@type": "Organization", name: "Lidr.io", url: SITE } as const;

export const postUrl = (slug: string) => `${SITE}/blog/${slug}/`;

export function blogPostingJsonLd(post: BuiltPost) {
  return {
    "@context": "https://schema.org",
    "@type": "BlogPosting",
    headline: post.title,
    description: post.description,
    image: absolute(post.ogImage),
    datePublished: post.date,
    dateModified: post.updated ?? post.date,
    author: ORG,
    publisher: { "@type": "Organization", name: "Lidr.io", logo: { "@type": "ImageObject", url: `${SITE}/images/Group-198.webp` } },
    mainEntityOfPage: post.canonical ?? postUrl(post.slug),
  };
}

export function breadcrumbJsonLd(post: BuiltPost, tag?: { slug: string; name: string }) {
  const items = [
    { name: "Blog", item: `${SITE}/blog/` },
    ...(tag ? [{ name: tag.name, item: `${SITE}/blog/tag/${tag.slug}/` }] : []),
    { name: post.title, item: postUrl(post.slug) },
  ];
  return {
    "@context": "https://schema.org",
    "@type": "BreadcrumbList",
    itemListElement: items.map((it, i) => ({ "@type": "ListItem", position: i + 1, ...it })),
  };

import type { Metadata } from "next";
import { notFound } from "next/navigation";
import { PostArticle } from "../../../components/blog/PostArticle";
import { getPost, getPosts, getTag, getTags, paramsOrPlaceholder, PLACEHOLDER } from "../../../lib/blog/data";
import { absolute, blogPostingJsonLd, breadcrumbJsonLd, postUrl } from "../../../lib/blog/jsonld";
import type { BuiltPost } from "../../../lib/blog/types";

type Props = { params: Promise<{ slug: string }> };

export const dynamicParams = false;

export function generateStaticParams() {
  return paramsOrPlaceholder("slug", getPosts().map((p) => p.slug));
}

export async function generateMetadata({ params }: Props): Promise<Metadata> {
  const { slug } = await params;
  const post = getPost(slug);
  if (!post) return {};
  const url = post.canonical ?? postUrl(post.slug);
  const image = { url: absolute(post.ogImage), width: post.cover?.width ?? 1200, height: post.cover?.height ?? 630, alt: post.cover?.alt ?? post.title };
  return {
    title: post.title,
    description: post.description,
    alternates: { canonical: url, types: { "application/rss+xml": "https://lidr.io/blog/rss.xml" } },
    openGraph: {
      type: "article",
      title: post.title,
      description: post.description,
      url,
      siteName: "Lidr.io",
      locale: "en_US",
      publishedTime: post.date,
      modifiedTime: post.updated ?? post.date,
      tags: post.tags,
      images: [image],
    },
    twitter: { card: "summary_large_image", title: post.title, description: post.description, images: [image.url] },
  };
}

export default async function PostPage({ params }: Props) {
  const { slug } = await params;
  if (slug === PLACEHOLDER) notFound();
  const post = getPost(slug);
  if (!post) notFound();
  const related = post.related.map((s) => getPost(s)).filter((p): p is BuiltPost => !!p);
  const primaryTag = post.tags[0] ? getTag(post.tags[0]) : undefined;
  const jsonLd = [blogPostingJsonLd(post), breadcrumbJsonLd(post, primaryTag)];
  return (
    <>
      <script type="application/ld+json" dangerouslySetInnerHTML={{ __html: JSON.stringify(jsonLd).replace(/</g, "\\u003c") }} />
      <PostArticle post={post} tags={getTags()} related={related} />
    </>
  );
}

import { getPosts } from "../../../lib/blog/data";
import { postUrl } from "../../../lib/blog/jsonld";

const esc = (s: string) =>
  s.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;");

export function GET() {

    .slice(0, 50)
    .map(

      <title>${esc(p.title)}</title>
      <link>${postUrl(p.slug)}</link>
      <guid isPermaLink="true">${postUrl(p.slug)}</guid>
      <pubDate>${new Date(`${p.date}T00:00:00Z`).toUTCString()}</pubDate>
      <description>${esc(p.description)}</description>
    </item>`
    )
    .join("\n");
  const xml = `<?xml version="1.0" encoding="UTF-8"?>
<rss version="2.0">
  <channel>
    <title>Lidr.io Blog</title>
    <link>https://lidr.io/blog/</link>
    <description>Guides for site teams on photo records and reporting.</description>
    <language>en-gb</language>
${items}
  </channel>
</rss>
`;
  return new Response(xml, { headers: { "Content-Type": "application/rss+xml; charset=utf-8" } });
}
